Fitting Linear Regression Models

For each outcome and binary characteristic, fit the model

characteristics <- c(
  "funny", "show_product_quickly", "patriotic",
  "celebrity", "danger", "animals", "use_sex"
)

# Convert logical values to 0 and 1.
youtube_clean[characteristics] <- lapply(
  youtube_clean[characteristics],
  as.integer
)

model_grid <- expand.grid(
  outcome = valid_outcomes,
  characteristic = characteristics,
  stringsAsFactors = FALSE
)

model_results <- do.call(rbind, lapply(
  seq_len(nrow(model_grid)),
  function(i) {
    outcome <- model_grid$outcome[i]
    characteristic <- model_grid$characteristic[i]

    # Include one characteristic and continuous year.
    model_formula <- reformulate(
      c(characteristic, "year"),
      response = paste0("log_", outcome)
    )

    # Missing observations are omitted separately for each model.
    fit <- lm(
      model_formula,
      data = youtube_clean,
      na.action = na.omit
    )

    coefficient_table <- summary(fit)$coefficients

    data.frame(
      outcome = outcome,
      characteristic = characteristic,
      estimate = coefficient_table[characteristic, "Estimate"],
      p_value = coefficient_table[characteristic, "Pr(>|t|)"],
      n = nobs(fit)
    )
  }
))

# Identify significant associations at the 0.05 level.
alpha <- 0.05

model_results$significant <- model_results$p_value < alpha
model_results$direction <- ifelse(
  model_results$estimate > 0,
  "positive",
  "negative"
)

# Report significant characteristics separately for each outcome.
significance_summary <- do.call(rbind, lapply(
  valid_outcomes,
  function(outcome) {
    selected <- subset(
      model_results,
      model_results$outcome == outcome & significant
    )

    associations <- if (nrow(selected) == 0) {
      "None"
    } else {
      paste(
        paste0(
          selected$characteristic,
          " (", selected$direction, ")"
        ),
        collapse = ", "
      )
    }

    data.frame(
      outcome = outcome,
      significant_associations = associations
    )
  }
))

Estimating Regression Coefficients Using Matrix Algebra

# Select the characteristic with the smallest view-count p-value.
best_characteristic <- view_results$characteristic[
  which.min(view_results$p_value)
]

print(best_characteristic)
[1] "danger"
# Use complete observations for the selected model
needed_columns <- c(
  "log_view_count", best_characteristic, "year"
)

model_data <- youtube_clean[
  complete.cases(youtube_clean[, needed_columns]),
  needed_columns
]

# Construct the design matrix:
X <- cbind(
  intercept = 1,
  characteristic = model_data[[best_characteristic]],
  year = model_data$year
)

colnames(X) <- c("(Intercept)", best_characteristic, "year")

# Construct the outcome vector.
Y <- model_data$log_view_count

# Calculate OLS coefficients
beta_matrix <- solve(t(X) %*% X) %*% t(X) %*% Y

print(beta_matrix)

# Fit the model
comparison_formula <- reformulate(
  c(best_characteristic, "year"),
  response = "log_view_count"
)

fit_lm <- lm(
  comparison_formula,
  data = model_data
)

# Compare the two sets of estimates
comparison <- data.frame(
  term = colnames(X),
  matrix_estimate = as.numeric(beta_matrix),
  lm_estimate = as.numeric(coef(fit_lm)),
  difference = as.numeric(beta_matrix) - as.numeric(coef(fit_lm))
)
